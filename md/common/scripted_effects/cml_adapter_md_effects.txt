# CML-Adapter für Millennium Dawn v2.0.3 (Spec 4.1): Effekte.
# Meinungs-Adapter lesen temp_opinion (Aufrufer setzt es); MD kopiert es selbst in
# temp_opinion_change (00_internal_faction_effects.txt, autocrats_opinion_change).

# Kader-Gruppe statt Rohstoffindustrie (Muster SOV_communist_purges, swap_ideas).
cml_adapter_install_cadres = {
	log = "[GetDateText]: [Root.GetName]: cml_adapter_install_cadres"
	if = {
		limit = { has_idea = fossil_fuel_industry }
		swap_ideas = {
			remove_idea = fossil_fuel_industry
			add_idea = communist_cadres
		}
	}
}

# Oligarchen raus, Gewerkschaften rein, danach Gewerkschaften +unions_after_swap.
cml_adapter_purge_oligarchs = {
	log = "[GetDateText]: [Root.GetName]: cml_adapter_purge_oligarchs"
	if = {
		limit = { has_idea = oligarchs }
		swap_ideas = {
			remove_idea = oligarchs
			add_idea = labour_unions
		}
		set_temp_variable = { temp_opinion = @@unions_after_swap@@ }
		change_labour_unions_opinion = yes
	}
}
